# Create ClickHouse Cluster VMs (3 nodes)
# Specs: 8 vCPU, 16GB RAM, 100GB disk each

## ========================================
## PREREQUISITES (one time setup)
## ========================================

cd /var/lib/libvirt/images

# Download Ubuntu 24.04 cloud image (if not exists)
wget https://cloud-images.ubuntu.com/noble/current/noble-server-cloudimg-amd64.img -O ubuntu-24.04-cloudimg.img

# Verify base image
ls -lh ubuntu-24.04-cloudimg.img

## ========================================
## CH-NODE-01 (192.168.150.140)
## ========================================

# 1. Create disk from base image
qemu-img create -f qcow2 -F qcow2 -b ubuntu-24.04-cloudimg.img ch-node-01.qcow2 100G

# 2. Create cloud-init config
SSH_KEY=$(cat /root/.ssh/okd-bootstrap-key.pub)
cat > /tmp/cloud-init-ch01.yaml << EOF
#cloud-config
hostname: ch-node-01
fqdn: ch-node-01.local
manage_etc_hosts: true
users:
  - name: root
    ssh_authorized_keys:
      - $SSH_KEY
    lock_passwd: false
chpasswd:
  list: |
    root:clickhouse123
  expire: False
ssh_pwauth: True
package_update: true
packages:
  - curl
  - wget
  - vim
  - htop
  - net-tools
runcmd:
  - systemctl restart ssh
EOF

# 3. Create cloud-init ISO
cloud-localds /var/lib/libvirt/images/ch-node-01-init.iso /tmp/cloud-init-ch01.yaml

# 4. Create VM
virt-install \
  --name ch-node-01 \
  --memory 16384 \
  --vcpus 8 \
  --disk path=/var/lib/libvirt/images/ch-node-01.qcow2,format=qcow2,bus=virtio \
  --disk path=/var/lib/libvirt/images/ch-node-01-init.iso,device=cdrom \
  --network network=okdnet,model=virtio,mac=52:54:00:aa:bb:01 \
  --os-variant ubuntu22.04 \
  --import \
  --noautoconsole

# 5. Enable autostart
virsh autostart ch-node-01

# 6. Check VM (wait 30-60 seconds)
virsh list | grep ch-node-01
virsh net-dhcp-leases okdnet | grep ch-node-01
ssh root@192.168.150.140 hostname
