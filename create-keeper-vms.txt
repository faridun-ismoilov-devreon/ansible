# Create ClickHouse Keeper VMs (3 nodes)
# Specs: 2 vCPU, 4GB RAM, 30GB disk each

cd /var/lib/libvirt/images

## ========================================
## KEEPER-01 (192.168.150.143)
## ========================================

# 1. Create disk
qemu-img create -f qcow2 -F qcow2 -b ubuntu-24.04-cloudimg.img keeper-01.qcow2 30G

# 2. Create cloud-init config
SSH_KEY=$(cat /root/.ssh/okd-bootstrap-key.pub)
cat > /tmp/cloud-init-keeper01.yaml << EOF
#cloud-config
hostname: keeper-01
fqdn: keeper-01.local
manage_etc_hosts: true
users:
  - name: root
    ssh_authorized_keys:
      - $SSH_KEY
    lock_passwd: false
chpasswd:
  list: |
    root:clickhouse123
  expire: False
ssh_pwauth: True
package_update: true
packages:
  - curl
  - wget
  - vim
  - htop
  - net-tools
runcmd:
  - systemctl restart ssh
EOF

# 3. Create cloud-init ISO
cloud-localds /var/lib/libvirt/images/keeper-01-init.iso /tmp/cloud-init-keeper01.yaml

# 4. Create VM
virt-install \
  --name keeper-01 \
  --memory 4096 \
  --vcpus 2 \
  --disk path=/var/lib/libvirt/images/keeper-01.qcow2,format=qcow2,bus=virtio \
  --disk path=/var/lib/libvirt/images/keeper-01-init.iso,device=cdrom \
  --network network=okdnet,model=virtio,mac=52:54:00:aa:bb:11 \
  --os-variant ubuntu22.04 \
  --import \
  --noautoconsole

# 5. Enable autostart
virsh autostart keeper-01
