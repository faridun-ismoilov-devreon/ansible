# Install and Configure ClickHouse Keeper Cluster
# 3 nodes: keeper-01/02/03 (192.168.150.143/144/145)

## ========================================
## STEP 1: Install ClickHouse Keeper on all 3 nodes
## ========================================

# Run on each keeper node (or use loop):
# for IP in 143 144 145; do ssh root@192.168.150.$IP "commands"; done

# Install on keeper-01
ssh root@192.168.150.143 << 'EOF'
apt-get update
apt-get install -y apt-transport-https ca-certificates curl gnupg
curl -fsSL https://packages.clickhouse.com/rpm/lts/repodata/repomd.xml.key | gpg --dearmor -o /usr/share/keyrings/clickhouse-keyring.gpg
echo "deb [signed-by=/usr/share/keyrings/clickhouse-keyring.gpg] https://packages.clickhouse.com/deb stable main" | tee /etc/apt/sources.list.d/clickhouse.list
apt-get update
apt-get install -y clickhouse-keeper
systemctl stop clickhouse-keeper
systemctl disable clickhouse-keeper
EOF

# Install on keeper-02
ssh root@192.168.150.144 << 'EOF'

# Verify installation
ssh root@192.168.150.143 "clickhouse-keeper --version"
ssh root@192.168.150.144 "clickhouse-keeper --version"
ssh root@192.168.150.145 "clickhouse-keeper --version"

## ========================================
## STEP 2: Configure keeper-01 (server_id=1)
## ========================================

ssh root@192.168.150.143 << 'EOF'
cat > /etc/clickhouse-keeper/keeper_config.xml << 'KEEPERCONF'
<clickhouse>
    <logger>
        <level>information</level>
        <log>/var/log/clickhouse-keeper/clickhouse-keeper.log</log>
        <errorlog>/var/log/clickhouse-keeper/clickhouse-keeper.err.log</errorlog>
        <size>1000M</size>
        <count>3</count>
    </logger>

    <listen_host>0.0.0.0</listen_host>

    <keeper_server>
        <tcp_port>9181</tcp_port>
        <server_id>1</server_id>
        
        <log_storage_path>/var/lib/clickhouse-keeper/coordination/log</log_storage_path>
        <snapshot_storage_path>/var/lib/clickhouse-keeper/coordination/snapshots</snapshot_storage_path>

        <coordination_settings>
            <operation_timeout_ms>10000</operation_timeout_ms>
            <session_timeout_ms>30000</session_timeout_ms>
            <raft_logs_level>information</raft_logs_level>
        </coordination_settings>

        <raft_configuration>
            <server>
                <id>1</id>
                <hostname>192.168.150.143</hostname>
                <port>9234</port>
            </server>
            <server>
                <id>2</id>
                <hostname>192.168.150.144</hostname>
                <port>9234</port>
            </server>
            <server>
                <id>3</id>
                <hostname>192.168.150.145</hostname>
                <port>9234</port>
            </server>
        </raft_configuration>
    </keeper_server>
</clickhouse>
KEEPERCONF

mkdir -p /var/lib/clickhouse-keeper/coordination/log
mkdir -p /var/lib/clickhouse-keeper/coordination/snapshots
chown -R clickhouse:clickhouse /var/lib/clickhouse-keeper
chown -R clickhouse:clickhouse /var/log/clickhouse-keeper
EOF

## ========================================
## STEP 5: Start Keeper Cluster
## ========================================

# Start all keeper nodes
ssh root@192.168.150.143 "systemctl enable clickhouse-keeper && systemctl start clickhouse-keeper"
ssh root@192.168.150.144 "systemctl enable clickhouse-keeper && systemctl start clickhouse-keeper"
ssh root@192.168.150.145 "systemctl enable clickhouse-keeper && systemctl start clickhouse-keeper"

# Wait for cluster to form
echo "Waiting 10 seconds for keeper cluster to form..."
sleep 10

## ========================================
## STEP 6: Verify Keeper Cluster
## ========================================

# Check services are running
ssh root@192.168.150.143 "systemctl status clickhouse-keeper --no-pager | head -3"
ssh root@192.168.150.144 "systemctl status clickhouse-keeper --no-pager | head -3"
ssh root@192.168.150.145 "systemctl status clickhouse-keeper --no-pager | head -3"

# Check keeper status (ruok = are you ok?)
echo "Checking keeper-01:"
echo "ruok" | nc 192.168.150.143 9181

echo "Checking keeper-02:"
echo "ruok" | nc 192.168.150.144 9181

echo "Checking keeper-03:"
echo "ruok" | nc 192.168.150.145 9181

# Check leader election (mntr = monitor)
echo "Keeper cluster info:"
echo "mntr" | nc 192.168.150.143 9181 | grep -E "zk_server_state|zk_followers|zk_synced_followers"

# If you see "imok" responses and one node is "leader", cluster is ready! ✅
