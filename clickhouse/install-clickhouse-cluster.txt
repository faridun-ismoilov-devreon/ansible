# Install ClickHouse Cluster (3 nodes + 3 keepers)
# Topology: 1 shard × 3 replicas

## ========================================
## STEP 1: Install ClickHouse on all 3 nodes
## ========================================

# Install on ch-node-01 (192.168.150.140)
ssh root@192.168.150.140 << 'EOF'
curl -fsSL https://packages.clickhouse.com/rpm/lts/repodata/repomd.xml.key | gpg --batch --yes --dearmor -o /usr/share/keyrings/clickhouse-keyring.gpg
echo "deb [signed-by=/usr/share/keyrings/clickhouse-keyring.gpg] https://packages.clickhouse.com/deb stable main" > /etc/apt/sources.list.d/clickhouse.list
apt-get update
DEBIAN_FRONTEND=noninteractive apt-get install -y clickhouse-server clickhouse-client
systemctl stop clickhouse-server
systemctl disable clickhouse-server
clickhouse-server --version
EOF

# Install on ch-node-02 (192.168.150.141)
ssh root@192.168.150.141 << 'EOF'
curl -fsSL https://packages.clickhouse.com/rpm/lts/repodata/repomd.xml.key | gpg --batch --yes --dearmor -o /usr/share/keyrings/clickhouse-keyring.gpg
echo "deb [signed-by=/usr/share/keyrings/clickhouse-keyring.gpg] https://packages.clickhouse.com/deb stable main" > /etc/apt/sources.list.d/clickhouse.list
apt-get update
DEBIAN_FRONTEND=noninteractive apt-get install -y clickhouse-server clickhouse-client
systemctl stop clickhouse-server
systemctl disable clickhouse-server
clickhouse-server --version
EOF

# Install on ch-node-03 (192.168.150.142)
ssh root@192.168.150.142 << 'EOF'
curl -fsSL https://packages.clickhouse.com/rpm/lts/repodata/repomd.xml.key | gpg --batch --yes --dearmor -o /usr/share/keyrings/clickhouse-keyring.gpg
echo "deb [signed-by=/usr/share/keyrings/clickhouse-keyring.gpg] https://packages.clickhouse.com/deb stable main" > /etc/apt/sources.list.d/clickhouse.list
apt-get update
DEBIAN_FRONTEND=noninteractive apt-get install -y clickhouse-server clickhouse-client
systemctl stop clickhouse-server
systemctl disable clickhouse-server
clickhouse-server --version
EOF

## ========================================
## STEP 2: Configure /etc/hosts on all nodes
## ========================================

# Add hostname resolution (required for replication)
for IP in 140 141 142; do
  ssh root@192.168.150.$IP "cat >> /etc/hosts" << 'HOSTSEOF'
192.168.150.140 ch-node-01 ch-node-01.local
192.168.150.141 ch-node-02 ch-node-02.local
192.168.150.142 ch-node-03 ch-node-03.local
192.168.150.143 keeper-01 keeper-01.local
192.168.150.144 keeper-02 keeper-02.local
192.168.150.145 keeper-03 keeper-03.local
HOSTSEOF
done

# Verify hostname resolution
ssh root@192.168.150.140 "ping -c 1 ch-node-02"

## ========================================
## STEP 3: Configure Keeper connection (same on all nodes)
## ========================================

# Create config.d directory on all nodes
ssh root@192.168.150.140 "mkdir -p /etc/clickhouse-server/config.d"
ssh root@192.168.150.141 "mkdir -p /etc/clickhouse-server/config.d"
ssh root@192.168.150.142 "mkdir -p /etc/clickhouse-server/config.d"

# Configure keeper connection (same on all 3 nodes)
# NOTE: Use only <zookeeper> section (not <keeper>) for ClickHouse 25.9+
for IP in 140 141 142; do
  ssh root@192.168.150.$IP "cat > /etc/clickhouse-server/config.d/keeper.xml" << 'KEEPERXML'
<clickhouse>
    <zookeeper>
        <node>
            <host>192.168.150.143</host>
            <port>9181</port>
        </node>
        <node>
            <host>192.168.150.144</host>
            <port>9181</port>
        </node>
        <node>
            <host>192.168.150.145</host>
            <port>9181</port>
        </node>
        <session_timeout_ms>30000</session_timeout_ms>
        <operation_timeout_ms>10000</operation_timeout_ms>
    </zookeeper>
</clickhouse>
KEEPERXML
done

## ========================================
## STEP 4: Configure cluster topology (same on all nodes)
## ========================================

for IP in 140 141 142; do
  ssh root@192.168.150.$IP "cat > /etc/clickhouse-server/config.d/remote_servers.xml" << 'CLUSTERXML'
<clickhouse>
    <remote_servers>
        <production_cluster>
            <shard>
                <internal_replication>true</internal_replication>
                <replica>
                    <host>192.168.150.140</host>
                    <port>9000</port>
                </replica>
                <replica>
                    <host>192.168.150.141</host>
                    <port>9000</port>
                </replica>
                <replica>
                    <host>192.168.150.142</host>
                    <port>9000</port>
                </replica>
            </shard>
        </production_cluster>
    </remote_servers>
    
    <distributed_ddl>
        <path>/clickhouse/task_queue/ddl</path>
    </distributed_ddl>
</clickhouse>
CLUSTERXML
done

## ========================================
## STEP 5: Configure macros (UNIQUE for each node)
## ========================================

# ch-node-01: replica=01
ssh root@192.168.150.140 "cat > /etc/clickhouse-server/config.d/macros.xml" << 'MACROSXML'
<clickhouse>
    <macros>
        <shard>01</shard>
        <replica>01</replica>
        <cluster>production_cluster</cluster>
    </macros>
</clickhouse>
MACROSXML

# ch-node-02: replica=02
ssh root@192.168.150.141 "cat > /etc/clickhouse-server/config.d/macros.xml" << 'MACROSXML'
<clickhouse>
    <macros>
        <shard>01</shard>
        <replica>02</replica>
        <cluster>production_cluster</cluster>
    </macros>
</clickhouse>
MACROSXML

# ch-node-03: replica=03
ssh root@192.168.150.142 "cat > /etc/clickhouse-server/config.d/macros.xml" << 'MACROSXML'
<clickhouse>
    <macros>
        <shard>01</shard>
        <replica>03</replica>
        <cluster>production_cluster</cluster>
    </macros>
</clickhouse>
MACROSXML

## ========================================
## STEP 6: Configure network and users
## ========================================

# Allow external connections (listen on all interfaces)
for IP in 140 141 142; do
  ssh root@192.168.150.$IP "cat > /etc/clickhouse-server/config.d/network.xml" << 'NETWORKXML'
<clickhouse>
    <listen_host>0.0.0.0</listen_host>
    <http_port>8123</http_port>
    <tcp_port>9000</tcp_port>
    <interserver_http_port>9009</interserver_http_port>
</clickhouse>
NETWORKXML
done

# Set permissions
ssh root@192.168.150.140 "chown -R clickhouse:clickhouse /etc/clickhouse-server"
ssh root@192.168.150.141 "chown -R clickhouse:clickhouse /etc/clickhouse-server"
ssh root@192.168.150.142 "chown -R clickhouse:clickhouse /etc/clickhouse-server"

## ========================================
## STEP 7: Start ClickHouse cluster
## ========================================

# Start all nodes
ssh root@192.168.150.140 "systemctl enable clickhouse-server && systemctl start clickhouse-server"
ssh root@192.168.150.141 "systemctl enable clickhouse-server && systemctl start clickhouse-server"
ssh root@192.168.150.142 "systemctl enable clickhouse-server && systemctl start clickhouse-server"

# Wait for services to start
echo "Waiting 15 seconds for ClickHouse to start..."
sleep 15

## ========================================
## STEP 8: Verify cluster
## ========================================

# Check services are running
ssh root@192.168.150.140 "systemctl status clickhouse-server --no-pager | head -3"
ssh root@192.168.150.141 "systemctl status clickhouse-server --no-pager | head -3"
ssh root@192.168.150.142 "systemctl status clickhouse-server --no-pager | head -3"

# Check ClickHouse version on all nodes
clickhouse-client --host 192.168.150.140 --query "SELECT version()"
clickhouse-client --host 192.168.150.141 --query "SELECT version()"
clickhouse-client --host 192.168.150.142 --query "SELECT version()"

# Check cluster configuration
echo "Cluster members:"
clickhouse-client --host 192.168.150.140 --query "SELECT cluster, shard_num, replica_num, host_name, port FROM system.clusters WHERE cluster = 'production_cluster' ORDER BY replica_num"

# Check keeper connection
echo "Keeper connection:"
clickhouse-client --host 192.168.150.140 --query "SELECT * FROM system.zookeeper WHERE path = '/'"

## ========================================
## STEP 9: Test replication
## ========================================

# Create test database
clickhouse-client --host 192.168.150.140 --query "CREATE DATABASE IF NOT EXISTS test ON CLUSTER production_cluster"

# Create replicated table
clickhouse-client --host 192.168.150.140 --query "
CREATE TABLE IF NOT EXISTS test.replication_test ON CLUSTER production_cluster
(
    id UInt64,
    message String,
    created_at DateTime DEFAULT now()
)
ENGINE = ReplicatedMergeTree('/clickhouse/tables/{shard}/test/replication_test', '{replica}')
ORDER BY id
"

# Insert data on ch-node-01
clickhouse-client --host 192.168.150.140 --query "INSERT INTO test.replication_test (id, message) VALUES (1, 'Test from ch-node-01')"

# Wait for replication
sleep 3

# Restart replicas on ch-node-02 and ch-node-03 to ensure sync
clickhouse-client --host 192.168.150.141 --query "SYSTEM RESTART REPLICA test.replication_test"
clickhouse-client --host 192.168.150.142 --query "SYSTEM RESTART REPLICA test.replication_test"

# Wait for sync
sleep 5

# Check data on all nodes
echo "Data on ch-node-01:"
clickhouse-client --host 192.168.150.140 --query "SELECT count(), max(id) FROM test.replication_test"

echo "Data on ch-node-02:"
clickhouse-client --host 192.168.150.141 --query "SELECT count(), max(id) FROM test.replication_test"

echo "Data on ch-node-03:"
clickhouse-client --host 192.168.150.142 --query "SELECT count(), max(id) FROM test.replication_test"

# If all nodes show count=1, replication works! ✅

# Test reverse replication (insert from ch-node-02)
clickhouse-client --host 192.168.150.141 --query "INSERT INTO test.replication_test (id, message) VALUES (2, 'Test from ch-node-02')"
sleep 3

echo "Verify data replicated to all nodes:"
clickhouse-client --host 192.168.150.140 --query "SELECT id, message FROM test.replication_test ORDER BY id"
clickhouse-client --host 192.168.150.142 --query "SELECT count() FROM test.replication_test"

# Cleanup test
clickhouse-client --host 192.168.150.140 --query "DROP TABLE IF EXISTS test.replication_test ON CLUSTER production_cluster"
